-- 1. TABLES
create table if not exists public.services (
  id bigint generated always as identity primary key,
  titre text not null, description text not null,
  icone text default '✦', actif boolean default true,
  ordre int default 0, created_at timestamptz default now()
);
create table if not exists public.laureats (
  id bigint generated always as identity primary key,
  nom text not null, legende text default 'Lauréats de la formation en anglais – EETC',
  photo_url text, created_at timestamptz default now()
);
create table if not exists public.messages (
  id bigint generated always as identity primary key,
  nom text not null, email text, telephone text,
  service text, message text not null,
  statut text default 'nouveau', created_at timestamptz default now()
);
create table if not exists public.settings (
  id int primary key, telephone text, email text, adresse text, updated_at timestamptz default now()
);

-- 3. RLS : lecture publique, écriture réservée aux admins connectés
alter table public.services enable row level security;
alter table public.laureats enable row level security;
alter table public.messages enable row level security;
alter table public.settings enable row level security;
-- Lecture publique
create policy "lecture publique services" on public.services for select using (true);
create policy "lecture publique laureats" on public.laureats for select using (true);
create policy "lecture publique settings" on public.settings for select using (true);
-- Tout le monde peut envoyer un message (formulaire public)
create policy "envoyer message" on public.messages for insert with check (true);
-- Admins connectés : tout gérer
create policy "admin services" on public.services for all using (auth.role() = 'authenticated') with check (auth.role() = 'authenticated');
create policy "admin laureats" on public.laureats for all using (auth.role() = 'authenticated') with check (auth.role() = 'authenticated');
create policy "admin messages" on public.messages for select using (auth.role() = 'authenticated');
create policy "admin messages delete" on public.messages for delete using (auth.role() = 'authenticated');
create policy "admin settings" on public.settings for all using (auth.role() = 'authenticated') with check (auth.role() = 'authenticated');
